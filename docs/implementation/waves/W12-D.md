export type DeadlineProcedure = 'HMK' | 'CMK' | 'İYUK' | 'İİK' | 'AYM' | 'Diğer';
export type DeadlineUnit = 'gun' | 'hafta' | 'ay' | 'yil';
export type DeadlineStartKind = 'teblig' | 'tefhim' | 'ogrenme' | 'karar';
export interface DeadlineRule {
  id: string; title: string; procedure: DeadlineProcedure;
  period: { value: number; unit: DeadlineUnit };        // value 0 when computable=false
  startKind: DeadlineStartKind;
  reference: { legislationNo: string; article: string; label: string };
  notes: string[]; adliTatilApplies: boolean;
  verified: { status: 'dogrulandi' | 'dogrulanmadi'; date: 'YYYY-MM-DD'; source: string };
  computable: boolean;                                   // additive
  periodLabel: string;                                   // additive, e.g. "2 hafta"
  transition?: { effectiveFrom: 'YYYY-MM-DD'; before: string; law: string }; // additive (CMK)
}

### Calculator (`control-plane/src/deadlines/calc.ts`)

```ts
export function computeDeadline(
  input: { ruleId?: string; custom?: { value: number; unit: DeadlineUnit }; startDate: 'YYYY-MM-DD'; applyAdliTatil?: boolean },
  options?: { rules?: readonly DeadlineRule[] },
): DeadlineComputation;   // throws DeadlineInputError { kind: 'INVALID_REQUEST' | 'RULE_NOT_COMPUTABLE', path?, message (Turkish) }

export interface DeadlineComputation {
  dueDate: 'YYYY-MM-DD'; dueWeekday: 'Pazartesi' | … | 'Pazar';
  rule: DeadlineRule | null; steps: string[]; warnings: string[]; disclaimer: string;
  // additive
  startDate: string; startDateTr: 'GG.AA.YYYY'; dueDateTr: 'GG.AA.YYYY';
  baseDueDate: string; baseDueDateTr: string;            // m.92 last day before tatil/roll
  period: { value; unit }; periodLabel: string;
  startKind: DeadlineStartKind | 'ozel'; procedure: DeadlineProcedure | 'ozel';
  adliTatil: { applied: boolean; baseInAdliTatil: boolean; dueDateWithoutExtension: string | null };
  holidayCalendarCovered: boolean;
}
export const CUSTOM_PERIOD_MAX = { gun: 3650, hafta: 520, ay: 120, yil: 10 };
```

Semantics (all deliberately conservative — earliest date, explained):

- Start day never counted; `N gün` → start + N; `N hafta` → start + 7N
  (same weekday); `N ay/yıl` → corresponding day, else last day of month.
- Roll-forward: Saturday, Sunday, 2429 fixed days, Ramazan/Kurban days.
  Arife / 28 Ekim stay working days + warning "işlemi öğleden önce tamamlayın".
- `applyAdliTatil` defaults to `rule.adliTatilApplies` (HMK computable rules
  and İYUK rules: true; CMK/İİK/AYM/Diğer: false; custom: false). When the
  m.92 last day is inside 20 Temmuz–31 Ağustos and the flag is on, the due
  date becomes 31 Ağustos + 7 = 7 Eylül, then rolled (07.09.2025 Pazar →
  08.09.2025). Warnings list HMK m.103 / İYUK m.61 exceptions with the
  un-extended date. A last day pushed into the tatil only by a weekend roll
  is NOT extended (disputed) — warned.
- CMK: never applies HMK m.104; when the period touches the tatil it emits
  the CMK m.331/4 warning (süreler işlemez + 3 gün; tutuklu işler excluded)
  and shows the earliest date. `applyAdliTatil=true` on a CMK rule is refused
  with a warning.
- Rules with `adliTatilApplies:false` can be forced with `applyAdliTatil:true`
  (warning "zorlandı"); HMK/İYUK rules can be switched off (warning names the
  extended date).
- Start-kind warnings: tebliğ (e-tebligat 5. gün, Tebligat K. m.7/a), tefhim
  (yoklukta tebliğ), öğrenme (ispat). CMK transition warning for start dates
  before 01.06.2024. Coverage warning for years without religious-holiday
  data. "İdari izin resmî tatil değildir" warning near a bayram.

### Router (`control-plane/src/deadlines/routes.ts`)

```
GET  /v1/deadlines/rules
  200 { rules: DeadlineRule[], disclaimer: string, adliTatil: { label: '20 Temmuz–31 Ağustos', basis: 'HMK m.102 / İYUK m.61' } }

GET  /v1/deadlines/holidays?year=YYYY            (additive endpoint)
  200 { year, holidays: { date, name, kind: 'ulusal'|'dini'|'yarim', source }[], adliTatil: { start, end, label }, religiousCalendarCovered: boolean, source, note, disclaimer }
  400 { error: { kind: 'INVALID_REQUEST', message, issues: [{ path: 'year', message }] } }

POST /v1/deadlines/compute
  body { ruleId?: string, custom?: { value: int>=1, unit: 'gun'|'hafta'|'ay'|'yil' }, startDate: 'YYYY-MM-DD', applyAdliTatil?: boolean }   (exactly one of ruleId/custom; .strict)
  200 DeadlineComputation
  400 { error: { kind: 'INVALID_REQUEST', message, issues: [{ path, message }] } }   — malformed body, unknown ruleId, both/neither, impossible date, custom bounds
  400 { error: { kind: 'RULE_NOT_COMPUTABLE', message, issues } }                    — hmk-islah, hmk-karar-duzeltme, is-arabuluculuk-dava-sarti
```

`createDeadlinesRouter(deps?: { rules?: readonly DeadlineRule[] }): Hono`.
No 5xx paths: the module has no I/O.
